# 授权登录


<details>
  <summary>client_id和client_secret的sql查询</summary>
  SELECT
  t.name,
  t.xiaoman_client_id,
  tt.client_type,
  t.xiaoman_user_id,
  u.email,
  t.client_id,
  t.secret,
  t.create_time,
  t.update_time,
  tt.valid_to,
  tt.status as 'client_status',
  p.enable_flag as 'crm_functional_okki'
FROM
  v5_account_base.tbl_oauth2_client_app t
  LEFT JOIN v5_account_base.tbl_client tt ON t.xiaoman_client_id = tt.client_id
  LEFT JOIN v5_account_base.tbl_user_info u ON t.xiaoman_user_id = u.user_id
  LEFT JOIN v5_account_base.tbl_client_privilege p ON t.xiaoman_client_id = p.client_id and p.privilege = 'crm.functional.okki'
 where xiaoman_client_id=14119
ORDER BY
  tt.client_type,
  t.xiaoman_client_id
limit 100;
