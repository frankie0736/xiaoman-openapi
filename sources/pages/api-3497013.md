```yaml
        1. 调用该接口会向 回调配置中 添加的 callback_url 发送一个请求

        1.1 Header 包含一个 X-Signature 校验头， X-Timestamp 时间戳

        2. 该接口需要一个正确的响应

        2.1 响应内容需要包含 message_id 这个与后续消息推送的响应一致

        2.2 测试接口需要额外返回biz_data 进行校验

        {"biz":[{"biz_type":"xiaoman.open.callback.test","biz_data":"xxx"}]}
        的数据结构

        2.3 响应体结构 ： {"code":200,"data":{"message_id":"msg_xxx","data":"xxx"}}


        JS 签名校验示例

        ```

        const crypto = require('crypto');


        // 生成 HMAC-SHA256 签名

        function generateHMAC(message, secretKey) {
            return crypto
                .createHmac('sha256', secretKey)
                .update(message)
                .digest('hex');
        }


        // 验证 HMAC-SHA256 签名

        function verifyHMAC(message, signature, secretKey) {
            const generatedSignature = generateHMAC(message, secretKey);
            return crypto.timingSafeEqual(
                Buffer.from(generatedSignature, 'hex'),
                Buffer.from(signature, 'hex')
            );
        }


        // 使用示例

        const message =
        '{"message_id":"msg_68d6527b8e41f8.33205216","data":"REDACTED_ENCRYPTED_DATA"}';
        // 接口返回的响应内容

        const secretKey = 'REDACTED_SECURITY_CONFIG'; //
        创建webhook 返回的 security_config

        const signature =
        'REDACTED_SIGNATURE'; //
        响应Header 上的 X-Signature

        const timestamp = '1758876283'; // 响应 Header 上的 X-Timestamp


        console.log(timestamp+message);

        // 验证签名

        const isValid = verifyHMAC(timestamp+message, signature, secretKey);

        console.log('Signature valid:', isValid);

        ```


        JS 解密示例

        ```

        const crypto = require('crypto');


        function decryptWithAES256GCM(data, key) {
            try {
                console.log('=== 解密调试信息 ===');
                console.log('输入data类型:', typeof data);
                console.log('输入key类型:', typeof key);
                
                // 确保输入数据正确
                let dataBuffer;
                if (typeof data === 'string') {
                    dataBuffer = Buffer.from(data, 'base64');
                } else {
                    dataBuffer = Buffer.isBuffer(data) ? data : Buffer.from(data);
                }
                
                console.log('dataBuffer长度:', dataBuffer.length);
                console.log('dataBuffer内容(HEX):', dataBuffer.toString('hex'));
                
                // 确保密钥正确处理
                let keyBuffer;
                if (typeof key === 'string') {
                    keyBuffer = Buffer.from(key, 'base64');
                } else {
                    keyBuffer = Buffer.isBuffer(key) ? key : Buffer.from(key);
                }
                
                console.log('keyBuffer长度:', keyBuffer.length);
                console.log('keyBuffer内容(HEX):', keyBuffer.toString('hex'));
                
                // 验证密钥长度
                if (keyBuffer.length !== 32) {
                    throw new Error(`密钥长度不正确: ${keyBuffer.length} 字节，应为32字节`);
                }
                
                // 检查数据长度
                if (dataBuffer.length < 28) {
                    throw new Error('加密数据太短');
                }
                
                // 提取IV、标签和加密内容
                const iv = dataBuffer.subarray(0, 12);
                const tag = dataBuffer.subarray(12, 28);
                const encrypted = dataBuffer.subarray(28);
                
                console.log('IV length:', iv.length);
                console.log('Tag length:', tag.length);
                console.log('Encrypted data length:', encrypted.length);
                console.log('IV (hex):', iv.toString('hex'));
                console.log('Tag (hex):', tag.toString('hex'));
                console.log('Encrypted (hex):', encrypted.toString('hex'));
                
                // 验证提取的数据完整性
                if (iv.length !== 12 || tag.length !== 16) {
                    throw new Error('加密数据格式不正确');
                }
                
                // 创建解密器
                const decipher = crypto.createDecipheriv('aes-256-gcm', keyBuffer, iv);
                
                // 设置认证标签
                decipher.setAuthTag(tag);
                
                // 解密数据
                let decrypted = decipher.update(encrypted, null, 'utf8');
                console.log('update阶段解密成功');
                decrypted += decipher.final('utf8');
                console.log('final阶段解密成功');
                
                return decrypted;
            } catch (error) {
                console.error('解密失败详细信息:', {
                    message: error.message,
                    stack: error.stack,
                    dataLength: data ? (typeof data === 'string' ? Buffer.from(data, 'base64').length : data.length) : 0
                });
                throw new Error(`解密失败: ${error.message}`);
            }
        }


        // 测试数据

        const message =
        '{"message_id":"msg_68d6527b8e41f8.33205216","data":"REDACTED_ENCRYPTED_DATA"}';

        const secretKey = 'REDACTED_SECURITY_CONFIG';


        console.log('原始消息:', message);

        console.log('密钥:', secretKey);


        // 解析消息

        const messageJson = JSON.parse(message);

        const messageData = messageJson.data;


        console.log('要解密的data字段:', messageData);


        try {
            const decryptedData = decryptWithAES256GCM(messageData, secretKey);
            console.log('解密成功:', decryptedData);
            
            // 尝试解析解密后的JSON
            try {
                const parsedData = JSON.parse(decryptedData);
                console.log('解析后的JSON:', JSON.stringify(parsedData, null, 2));
            } catch (parseError) {
                console.log('解密数据不是有效的JSON格式');
                console.log('解密数据内容:', decryptedData);
            }
        } catch (error) {
            console.error('解密失败:', error.message);
            
            // 额外的调试信息
            console.log('\n=== 额外调试信息 ===');
            console.log('Base64解码后的数据长度:', Buffer.from(messageData, 'base64').length);
            console.log('Base64数据是否有效:', /^[A-Za-z0-9+/]*={0,2}$/.test(messageData));
        }
