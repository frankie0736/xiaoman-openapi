# 创建订阅配置

## OpenAPI Specification

```yaml
openapi: 3.0.1
info:
  title: ''
  description: ''
  version: 1.0.0
paths:
  /v1/subscribe:
    post:
      summary: 创建订阅配置
      deprecated: false
      description: >+
        ## 请求参数说明

        1. subscribe_type ：订阅的模块

        2. subscribe_configs ：订阅的配置

        3. biz_type：订阅的事件

        特殊说明： 一次只能订阅一个模块，但可以同时订阅一个模块下的多个事件

        ## 支持的订阅类型

        | 模块 | 创建事件 | 更新事件 | 删除事件 |返回字段|

        |----------------|-----------------|-----------------|-----------------|---------------------|

        | crm.order | xiaoman.crm.order.create | xiaoman.crm.order.update |
        xiaoman.crm.order.delete | order_id, order_no |

        | crm.company | xiaoman.crm.company.create | xiaoman.crm.company.update
        | xiaoman.crm.company.delete | company_id, serial_id |

        | crm.dynamic_trail | xiaoman.crm.dynamic_trail.create |
        xiaoman.crm.dynamic_trail.update | xiaoman.crm.dynamic_trail.delete |
        trail_id |

        | crm.purchase_order | xiaoman.crm.purchase_order.create |
        xiaoman.crm.purchase_order.update | xiaoman.crm.purchase_order.delete |
        purchase_order_id, purchase_order_no |

        | crm.product | xiaoman.crm.product.create | xiaoman.crm.product.update
        | xiaoman.crm.product.delete | product_id, product_no |

        | crm.cash_collection_invoice |
        xiaoman.crm.cash_collection_invoice.create |
        xiaoman.crm.cash_collection_invoice.update |
        xiaoman.crm.cash_collection_invoice.delete | cash_collection_invoice_id,
        cash_collection_invoice_no |

        | crm.cash_collection | xiaoman.crm.cash_collection.create |
        xiaoman.crm.cash_collection.update | xiaoman.crm.cash_collection.delete
        | cash_collection_id |

        | crm.payment_invoice | xiaoman.crm.payment_invoice.create |
        xiaoman.crm.payment_invoice.update | xiaoman.crm.payment_invoice.delete
        | payment_invoice_id, payment_invoice_no |

        | crm.cost_invoice | xiaoman.crm.cost_invoice.create |
        xiaoman.crm.cost_invoice.update | xiaoman.crm.cost_invoice.delete |
        cost_invoice_id, cost_invoice_no |

        | crm.quotation | xiaoman.crm.quotation.create |
        xiaoman.crm.quotation.update | xiaoman.crm.quotation.delete |
        quotation_id, quotation_no |

        | crm.supplier | xiaoman.crm.supplier.create |
        xiaoman.crm.supplier.update | xiaoman.crm.supplier.delete | supplier_id,
        supplier_no |

        | crm.opportunity | xiaoman.crm.opportunity.create |
        xiaoman.crm.opportunity.update | xiaoman.crm.opportunity.delete |
        opportunity_id, serial_id |

        | crm.warehouse | xiaoman.crm.warehouse.create |
        xiaoman.crm.warehouse.update | xiaoman.crm.warehouse.delete |
        warehouse_id, warehouse_no |

        | crm.lead | xiaoman.crm.lead.create | xiaoman.crm.lead.update |
        xiaoman.crm.lead.delete | lead_id, serial_id |

        | crm.outbound_invoice | xiaoman.crm.outbound_invoice.create |
        xiaoman.crm.outbound_invoice.update |
        xiaoman.crm.outbound_invoice.delete | outbound_invoice_id, serial_id |

        | crm.inbound_invoice | xiaoman.crm.inbound_invoice.create |
        xiaoman.crm.inbound_invoice.update | xiaoman.crm.inbound_invoice.delete
        | inbound_invoice_id, serial_id |

        | crm.warehouse_return_invoice |
        xiaoman.crm.warehouse_return_invoice.create |
        xiaoman.crm.warehouse_return_invoice.update |
        xiaoman.crm.warehouse_return_invoice.delete |
        warehouse_return_invoice_id, serial_id |

      tags:
        - 消息推送
      parameters: []
      requestBody:
        content:
          application/json:
            schema:
              type: object
              properties: {}
            example:
              subscribe_type: crm.order
              subscribe_configs:
                - biz_type: xiaoman.crm.order.create
      responses:
        '200':
          description: ''
          content:
            application/json:
              schema:
                type: object
                properties:
                  code:
                    type: integer
                  message:
                    type: string
                  now:
                    type: string
                  data:
                    type: array
                    items:
                      type: string
                required:
                  - code
                  - message
                  - now
                  - data
                x-apifox-orders:
                  - code
                  - message
                  - now
                  - data
              example:
                code: 200
                message: success
                now: '2025-09-25 12:59:21'
                data: []
          headers: {}
          x-apifox-name: 成功
      security: []
      x-apifox-folder: 消息推送
      x-apifox-status: released
      x-run-in-apifox: https://open.xiaoman.cn/web/project/345096/apis/api-3497012-run
components:
  schemas: {}
  securitySchemes: {}
servers:
  - url: https://api-sandbox.xiaoman.cn
    description: 正式环境--14833的账户
security: []
```
