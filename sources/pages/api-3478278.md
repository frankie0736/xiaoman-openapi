# 回款单新建/编辑

## OpenAPI Specification

```yaml
openapi: 3.0.1
info:
  title: ''
  description: ''
  version: 1.0.0
paths:
  /v1/invoices/receipt/push:
    post:
      summary: 回款单新建/编辑
      deprecated: false
      description: ''
      tags:
        - 回款单
      parameters:
        - name: Authorization
          in: header
          description: 从/v1/oauth2/access_token获取到的access_token
          required: true
          example: '{{access_token}}'
          schema:
            type: string
            default: '{{access_token}}'
      requestBody:
        content:
          application/json:
            schema:
              type: object
              properties:
                '12345678':
                  type: string
                  description: >-
                    自定义字段，若自定义单选字段，值来自/v1/invoices/order/fields中对应字段的ext_info中的一个或多个
                order_id:
                  description: 如果关联的是订单，则设置order_id,不要设置opportunity_id
                  type: integer
                collection_date:
                  description: 回款日期
                  type: string
                amount:
                  type: integer
                  description: 回款金额(打款金额)
                currency:
                  type: string
                  description: 币种
                exchange_rate:
                  type: number
                  description: 当前币种换人民币汇率，不传取系统取系统设置的汇率，与exchange_rate_usd可以同时传，但必须有其中一个
                finance_verify_date:
                  description: 财务确认时间，可选字段(该字段废弃，请勿使用)
                  type: string
                comment:
                  description: 备注，可选字段
                  type: string
                payee:
                  description: 财务确认人，可选
                  type: string
                collect_status:
                  type: integer
                  description: 回款状态，值为 1 已生效，0 未生效，默认0
                trade_no:
                  type: string
                  description: 交易号/票据号
                cash_collection_no:
                  type: string
                  description: 回款单号
                exchange_rate_usd:
                  type: number
                  description: 当前币种换美元汇率，不传系统取系统配置的汇率，与exchange_rate可以同时传，但必须有其中一个
                finance_verify_name:
                  type: string
                  description: 财务确认人，可选字段(该字段废弃，请勿使用)
                capital_account_id:
                  type: integer
                  description: 资金账户，值传入资金账户ID，/v1/capitalAccount/list. capital_account_id
                bank_charge:
                  type: number
                  description: 回款手续费
                file_list:
                  type: array
                  items:
                    type: string
                    description: 回款单附件，传入可访问的文件url
              required:
                - amount
                - currency
              x-apifox-orders:
                - order_id
                - collection_date
                - amount
                - currency
                - exchange_rate
                - exchange_rate_usd
                - bank_charge
                - finance_verify_name
                - finance_verify_date
                - comment
                - payee
                - collect_status
                - trade_no
                - cash_collection_no
                - capital_account_id
                - file_list
                - '12345678'
            example: |-
              {
                   "order_id":1100025081, // 如果关联的是订单，则设置order_id,不要设置opportunity_id
                   //"opportunity_id":3333333 // 如果关联是商机，则设置opportunity_id，不要设置order_id
                   "collection_date":"2019-11-20", // 回款日期
                   "amount":100,"type":"L/C", // /v1/invoices/receipt/types 接口返回的付款类型
                   "currency":"CNY", // 金额
                   "exchange_rate":15, // 汇率
                   "finance_verify_name" : "李四", // 财务确认人，可选字段
                   "finance_verify_date":"2019-12-01" // 财务确认时间，可选字段
                   "comment":"这是18号货款", // 备注，可选字段
                   "payee": "张三" // 收款人，可选字段
                   "1101054282":'已汇款' //自定义字段
              }
      responses:
        '200':
          description: ''
          content:
            application/json:
              schema:
                type: object
                properties:
                  code:
                    type: integer
                  message:
                    type: string
                  now:
                    type: string
                  data:
                    type: object
                    properties:
                      cash_collection_id:
                        type: integer
                        title: 回款单ID
                      cash_collection_no:
                        type: string
                        title: 回款单编号
                    required:
                      - cash_collection_id
                      - cash_collection_no
                    x-apifox-orders:
                      - cash_collection_id
                      - cash_collection_no
                required:
                  - code
                  - message
                  - now
                  - data
                x-apifox-orders:
                  - code
                  - message
                  - now
                  - data
              example:
                code: 200
                message: success
                now: '2024-05-20 18:01:15'
                data:
                  cash_collection_id: 3555544681
                  cash_collection_no: HKD2024052001568
          headers: {}
          x-apifox-name: 成功
      security: []
      x-apifox-folder: 回款单
      x-apifox-status: released
      x-run-in-apifox: https://open.xiaoman.cn/web/project/345096/apis/api-3478278-run
components:
  schemas: {}
  securitySchemes: {}
servers:
  - url: https://api-sandbox.xiaoman.cn
    description: 正式环境--14833的账户
security: []
```
