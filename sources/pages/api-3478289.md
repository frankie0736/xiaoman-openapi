openapi: 3.0.1
info:
  title: ''
  description: ''

paths:
  /v1/opportunity/push:
    post:
      summary: 商机新建/编辑接口
      deprecated: false

        | 异常码 | 异常消息 |异常解决方案|

        | --- | --- | --- |

      tags:
        - 商机
        - s7.1
      parameters:
        - name: Authorization
          in: header
          description: 从/v1/oauth2/access_token获取到的access_token
          required: true
          example: '{{access_token}}'
          schema:
            type: string
            default: '{{access_token}}'
      requestBody:
        content:
          application/json:
            schema:
              type: object
              properties:
                opportunity_id:
                  type: integer
                  title: 商机ID
                  description: 编辑时必填  值来自/v1/opportunity/list 接口
                name:
                  type: string
                  title: 商机名称
                amount:
                  type: string
                  title: 金额
                exchange_rate:
                  type: string
                  title: 汇率
                exchange_rate_usd:
                  type: string
                  title: 美元汇率
                account_date:
                  type: string
                  title: 结束日期
                stage:
                  type: integer
                  title: 销售阶段ID
                  description: 值来自于/v1/opportunity/fields/selector 接口
                currency:
                  type: string
                  title: 币种
                company_id:
                  type: integer
                  title: 客户ID
                  description: 新建时必填 值来自/v1/company/list 接口
                customer_id:
                  type: array
                  items:
                    type: integer
                    description: 联系人ID 新建时必填 值来自/v1/company/list 接口，联系人是属于该客户下的customer_id
                  title: 联系人ID 数组
                origin_list:
                  type: array
                  items:
                    type: integer
                  title: 商机来源ID 数组
                fail_type:
                  type: integer
                  title: 输单原因ID
                  description: 值来自于/v1/opportunity/fields/selector 接口
                fail_remark:
                  type: string
                  title: 输单描述
                handler:
                  type: array
                  items:
                    type: integer
                    description: 协同跟进人ID
                  title: 协同跟进人
                  description: 值来自于/v1/user/list 接口
                main_user:
                  type: integer
                  title: 主负责人
                flow_id:
                  type: number
                  title: 销售流程ID
                  description: 值来自于/v1/opportunity/fields/selector 接口
                busines_type_id:
                  type: number
                  title: 业务类型
              required:
                - name
                - company_id
                - handler
                - busines_type_id
              x-apifox-orders:
                - opportunity_id
                - name
                - company_id
                - customer_id
                - amount
                - exchange_rate
                - exchange_rate_usd
                - account_date
                - origin_list
                - stage
                - currency
                - fail_type
                - fail_remark
                - handler
                - main_user
                - flow_id
                - busines_type_id
            example: |-
              {
                   "opportunity_id": "test",//有商机ID为更新操作、没有商机ID为创建商机操作
                      "name": "test",
                      "amount": "100000",
                      "exchange_rate": "700",
                      "exchange_rate_usd": "100",

              }

              schema:
                type: object
                properties:
                  code:
                    type: integer
                  message:
                    type: string
                  now:
                    type: string
                  data:
                    type: object
                    properties:
                      opportunity_id:
                        type: integer
                        title: 商机ID
                    required:
                      - opportunity_id
                    x-apifox-orders:
                      - opportunity_id
                required:
                  - code
                  - message
                  - now
                  - data
                x-apifox-orders:
                  - code
                  - message
                  - now
                  - data

                message: success
                now: '2025-03-07 17:03:25'
                data:
                  opportunity_id: 5203694883
          headers: {}

      security: []
      x-apifox-folder: 商机
      x-apifox-status: released
      x-run-in-apifox: https://open.xiaoman.cn/web/project/345096/apis/api-3478289-run
components:
  schemas: {}
  securitySchemes: {}

  - url: https://api-sandbox.xiaoman.cn
